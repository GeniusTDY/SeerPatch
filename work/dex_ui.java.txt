    private SpannableString dexHighlight(String text, String kw) {
        SpannableString ss = new SpannableString(text == null ? "" : text);
        if (kw == null || kw.length() == 0) return ss;
        String low = ss.toString().toLowerCase(Locale.ROOT);
        String k = kw.toLowerCase(Locale.ROOT);
        int from = 0;
        while (from <= low.length() - k.length()) {
            int i = low.indexOf(k, from);
            if (i < 0) break;
            ss.setSpan(new ForegroundColorSpan(C_GOLD), i, i + k.length(),
                    Spanned.SPAN_EXCLUSIVE_EXCLUSIVE);
            from = i + k.length();
        }
        return ss;
    }

    /** 描述命中时截取关键词附近的片段，避免整段描述挤满列表。 */
    private String dexSnippet(String desc, String kw, int radius) {
        if (desc == null || desc.length() == 0 || kw == null || kw.length() == 0) return "";
        String low = desc.toLowerCase(Locale.ROOT);
        int i = low.indexOf(kw.toLowerCase(Locale.ROOT));
        if (i < 0) return "";
        int start = Math.max(0, i - radius);
        int end = Math.min(desc.length(), i + kw.length() + radius);
        return (start > 0 ? "…" : "") + desc.substring(start, end)
                + (end < desc.length() ? "…" : "");
    }

    private static int dexTypeRank(String t) {
        if (t == null || t.length() == 0) return 900;
        if ("未知".equals(t)) return 999;
        for (int i = 0; i < DEX_TYPE_ORDER.length; i++) {
            if (DEX_TYPE_ORDER[i].equals(t)) return i;
        }
        for (int i = 0; i < DEX_TYPE_ORDER.length; i++) {
            if (t.startsWith(DEX_TYPE_ORDER[i])) return 100 + i;
        }
        return 800;
    }

    /** 收集并排序属性筛选项，首项固定为“全部”。 */
    private List<String> dexTypes(JSONArray src, String key) {
        List<String> out = new ArrayList<>();
        out.add("全部");
        if (src != null) {
            for (int i = 0; i < src.length(); i++) {
                JSONObject o = src.optJSONObject(i);
                if (o == null) continue;
                String t = o.optString(key, "未知");
                if (!out.contains(t)) out.add(t);
            }
        }
        Collections.sort(out.subList(1, out.size()), new Comparator<String>() {
            public int compare(String a, String b) {
                int ra = dexTypeRank(a);
                int rb = dexTypeRank(b);
                return ra != rb ? ra - rb : a.compareTo(b);
            }
        });
        return out;
    }

                String label = origin == skillList ? "← 返回技能列表" : "← 返回精灵";
                buildSkillDetail(skillDetail, sk, label, new Restore(origin), popen[0]);
                skillDetail.scrollTo(0, 0);
                detailBox.setVisibility(View.VISIBLE);
                skillDetail.setVisibility(View.VISIBLE);
                hideKeyboard(search);
            }
        };

        View.OnClickListener tabClick = new View.OnClickListener() {
            public void onClick(View v) {
                int m = v == tabSkill ? 1 : 0;
                if (mode[0] == m) return;
                mode[0] = m;
                search.setText("");
                search.setHint(m == 0 ? "搜索精灵名 / 编号" : "搜索技能名 / 描述 / ID");
                paintTabs.run();
                applyMode[0].run();
            }
        };
        tabPet.setOnClickListener(tabClick);
        tabSkill.setOnClickListener(tabClick);

        search.addTextChangedListener(new TextWatcher() {
            public void beforeTextChanged(CharSequence s, int a, int b, int c) {
            }

            public void onTextChanged(CharSequence s, int a, int b, int c) {
            }

            public void afterTextChanged(Editable s) {
                if (refresh[0] != null) refresh[0].run();
            }
        });
        search.setOnEditorActionListener(new TextView.OnEditorActionListener() {
            public boolean onEditorAction(TextView v, int actionId, KeyEvent event) {
                if (actionId == EditorInfo.IME_ACTION_SEARCH) {
                    hideKeyboard(search);
                    return true;
                }
                return false;
            }
        });

        paintTabs.run();
        applyMode[0].run();

        dlg.setContentView(root);
        Window window = dlg.getWindow();
        if (window != null) {
            window.setBackgroundDrawable(new ColorDrawable(0xF0040A17));
            window.setSoftInputMode(WindowManager.LayoutParams.SOFT_INPUT_ADJUST_RESIZE);
        }
        dlg.show();
        Window shown = dlg.getWindow();
        if (shown != null) {
            shown.setLayout(WindowManager.LayoutParams.MATCH_PARENT,
                    WindowManager.LayoutParams.MATCH_PARENT);
        }
    }

    /** 技能详情：标签、基础数据、客户端描述、补充说明、可学精灵。 */
    private void buildSkillDetail(ScrollView sv, JSONObject sk, String backLabel,
                                  final Runnable back, final PetOpen petOpen) {
        sv.removeAllViews();
        final int sid = sk.optInt("i");
        final String nm = sk.optString("n");
        final String ty = sk.optString("t", "未知");
        final String cat = dexCategory(sk.optInt("c"));
        final int col = dexTypeColor(ty);
        final int power = sk.optInt("p");
        final int acc = sk.optInt("a");
        final int pp = sk.optInt("pp");
        final int pri = sk.optInt("pr");
        final int cr = sk.optInt("cr");
        final int mon = sk.optInt("mo");
        final boolean stone = sid >= 1100001 && sid <= 1100080;
        final String desc = sk.optString("d", "");
        final String extra = sk.optString("x", "");
        List<int[]> learners = skillLearners == null ? null : skillLearners.get(sid);
        final int total = learners == null ? 0 : learners.size();

        LinearLayout col0 = new LinearLayout(this);
        col0.setOrientation(LinearLayout.VERTICAL);
        col0.setPadding(dp(2), 0, dp(2), dp(12));
        sv.addView(col0, new FrameLayout.LayoutParams(-1, -2));
        col0.addView(dexBackView(backLabel, back));

        LinearLayout hero = new LinearLayout(this);
        hero.setOrientation(LinearLayout.VERTICAL);
        hero.setGravity(Gravity.CENTER_HORIZONTAL);
        TextView name = new TextView(this);
        name.setText(nm);
        name.setTextSize(21);
        name.setTypeface(Typeface.DEFAULT_BOLD);
        name.setTextColor(C_TEXT);
        name.setGravity(Gravity.CENTER);
        hero.addView(name);

        LinearLayout tags = dexChipLine();
        tags.setPadding(0, dp(8), 0, 0);
        dexAddChip(tags, dexChipView(ty, col, (col & 0x00FFFFFF) | 0x22000000,
                (col & 0x00FFFFFF) | 0x66000000), 6);
        dexAddChip(tags, dexChipView(cat, C_CYAN, 0x145FD8F0, 0x665FD8F0), 6);
        if (sk.optInt("mh") == 1) {
            dexAddChip(tags, dexChipView("必中", C_CYAN, 0x145FD8F0, 0x665FD8F0), 6);
        }
        if (cr > 0) {
            dexAddChip(tags, dexChipView("致命一击 +" + cr + "/16",
                    C_GOLD, 0x1AF0C24B, 0x66F0C24B), 6);
        }
        if (stone) {
            dexAddChip(tags, dexChipView("技能石", 0xFFC9A6FF, 0x1AC9A6FF, 0x66C9A6FF), 6);
        }
        if (mon > 0) {
            dexAddChip(tags, dexChipView("BOSS", 0xFFFFB36B, 0x1AFFB36B, 0x66FFB36B), 0);
        }
        hero.addView(tags);

        TextView sub = new TextView(this);
        sub.setTextSize(10);
        sub.setTextColor(C_TEXT_FAINT);
        sub.setGravity(Gravity.CENTER);
        sub.setPadding(0, dp(6), 0, 0);
        sub.setText("ID " + sid + "   ·   可学精灵 " + total + " 只");
        hero.addView(sub);
        col0.addView(hero);

        LinearLayout statCard = dexCard();
        statCard.addView(dexSectionTitle("基础数据"));
        String powerText = power >= 100000 ? "占位" : (power > 0 ? String.valueOf(power) : "—");
        String accText = acc > 0 ? acc + "%" : "—";
        String ppText = pp > 0 ? String.valueOf(pp) : "—";
        String priText = pri > 0 ? "+" + pri : String.valueOf(pri);
        statCard.addView(dexStatGrid(new String[]{"威力", "命中", "PP", "先制"},
                new String[]{powerText, accText, ppText, priText}, null));
        dexAddCard(col0, statCard, 14);

        LinearLayout descCard = dexCard();
        descCard.addView(dexSectionTitle("技能描述"));
        TextView body = new TextView(this);
        body.setTextSize(12);
        body.setTextColor(desc.length() > 0 ? C_TEXT : C_TEXT_FAINT);
        body.setLineSpacing(dp(4), 1f);
        body.setPadding(0, dp(7), 0, dp(2));
        body.setText(desc.length() > 0 ? desc : "客户端未提供该技能的描述文本。");
        descCard.addView(body);
        dexAddCard(col0, descCard, 10);

        if (extra.length() > 0) {
            LinearLayout extraCard = new LinearLayout(this);
            extraCard.setOrientation(LinearLayout.VERTICAL);
            extraCard.setBackground(roundStroke(0x1AF0C24B, 0x66F0C24B, 12, 1));
            extraCard.setPadding(dp(12), dp(10), dp(12), dp(10));
            TextView label = new TextView(this);
            label.setText("补充说明");
            label.setTextSize(11);
            label.setTypeface(Typeface.DEFAULT_BOLD);
            label.setTextColor(C_GOLD);
            extraCard.addView(label);
            TextView text = new TextView(this);
            text.setTextSize(12);
            text.setTextColor(0xFFF6D98E);
            text.setLineSpacing(dp(4), 1f);
            text.setPadding(0, dp(6), 0, dp(1));
            text.setText(extra);
            extraCard.addView(text);
            dexAddCard(col0, extraCard, 10);
        }

        LinearLayout learnCard = dexCard();
        learnCard.addView(dexSectionTitle("可学精灵" + (total > 0 ? "（" + total + "）" : "")));
        if (total == 0) {
            TextView empty = new TextView(this);
            empty.setText(stone ? "技能石技能，无精灵通过升级习得" : "无精灵可通过升级习得该技能");
            empty.setTextSize(11);
            empty.setTextColor(C_TEXT_FAINT);
            empty.setPadding(0, dp(10), 0, dp(4));
            learnCard.addView(empty);
        } else {
            int cap = Math.min(total, 60);
            for (int i = 0; i < cap; i++) {
                int[] e = learners.get(i);
                JSONObject pet = dexPets.optJSONObject(e[0]);
                if (pet != null) learnCard.addView(dexLearnerRow(pet, e[1], sv, petOpen));
            }
            if (total > cap) {
                TextView more = new TextView(this);
                more.setText("仅显示前 60 只 · 共 " + total + " 只");
                more.setTextSize(9);
                more.setTextColor(C_TEXT_FAINT);
                more.setGravity(Gravity.CENTER);
                more.setPadding(0, dp(8), 0, dp(2));
                learnCard.addView(more);
            }
        }
        dexAddCard(col0, learnCard, 10);
    }
